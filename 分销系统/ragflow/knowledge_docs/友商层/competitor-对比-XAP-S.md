| 400 | 0.138 | 0.144 | -0.006 |
| 630 | 0.097 | 0.105 | -0.008 |
| 800 | 0.074 | 0.082 | -0.008 |
| 1000 | 0.056 | 0.064 | -0.008 |
| 1250 | 0.042 | 0.048 | -0.006 |
| 1600 | 0.032 | 0.036 | -0.004 |
| 2000 | 0.025 | 0.027 | -0.002 |
| 2500 | 0.019 | 0.024 | -0.005 |
| 3200 | 0.015 | 0.018 | -0.003 |
| 4000 | 0.012 | 0.013 | -0.001 |
